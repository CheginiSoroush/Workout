     6→export interface ExerciseVariant {
     7→  faName: string;
     8→  hevyName: string;
     9→  note: string;
    10→}
    11→
    12→export type ExerciseMode = "machine" | "free";
    13→
    14→export interface Exercise {
    15→  id: string;
    16→  hevyName: string; // نام دقیق در کتابخانه Hevy
    17→  faName: string; // نام فارسی
    18→  muscles: string; // عضلات هدف
    19→  sets: number;
    20→  reps: string;
    21→  rest: string;
    22→  rpe: string;
    23→  image: string; // مسیر عکس
    24→  videoQuery: string; // کوئری یوتیوب برای ویدیو آموزش
    25→  tips: string[]; // نکات فرم صحیح
    26→  mistakes: string[]; // خطاهای رایج
    27→  fst7?: boolean; // حرکت FST-7 (۷ ست پیاپی)
    28→
    29→  // ---- نسخه‌های جایگزین: دستگاه ↔ وزنه آزاد ----
    30→  // حرکت پایه کدام نسخه است؟ (پیش‌فرض: machine یعنی دستگاه/کابل)
    31→  baseMode?: ExerciseMode;
    32→  // نسخه دستگاهی جایگزین — وقتی حرکت اصلی وزنه آزاد است
    33→  machineAlt?: ExerciseVariant;
    34→  // نسخه وزنه آزاد جایگزین — وقتی حرکت اصلی دستگاه است
    35→  freeAlt?: ExerciseVariant;
    36→  // نسخه پیشنهادی برای شروع تازه‌کار (پیش‌فرض: baseMode)
    37→  startMode?: ExerciseMode;
    38→}
    39→
    40→export interface WorkoutDay {
    41→  id: string; // day1 | day2 | day3
    42→  weekDay: string; // شنبه / دوشنبه / چهارشنبه
    43→  title: string;
    44→  subtitle: string;
    45→  image: string;
    46→  duration: string;
    47→  why: string; // چرا این روز برای فرم شناگری مهمه
    48→  cardio: string;
    49→  exercises: Exercise[];
    50→}
